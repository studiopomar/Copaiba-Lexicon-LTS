    def duration_ms(self):
        return int(round(max(0.0, self._audio_dur * 1000.0)))

    @property

    def get_segment_times_ms(self) -> tuple[int, int]:
        if self._current_entry is None:
            return 0, 0

        e = self._current_entry
        dur_ms = self.duration_ms()

        offset_ms = int(e.offset)
        cutoff_ms = int(e.cutoff)

        start_ms = max(0, offset_ms)
        if cutoff_ms < 0:
            end_ms = dur_ms + cutoff_ms
        else:
            end_ms = cutoff_ms

        end_ms = min(end_ms, dur_ms)

        return start_ms, end_ms

    def get_mouse_sector(self) -> str:
        if self._current_entry is None:
            return "none"

        t_ms = self.last_mouse_ms()
        e = self._current_entry
        dur_ms = self.duration_ms()

        offset_ms = int(e.offset)
        overlap_ms = offset_ms + int(e.overlap)
        preutter_ms = offset_ms + int(e.preutter)
        consonant_ms = offset_ms + int(e.consonant)

        if e.cutoff < 0:
            cutoff_ms = dur_ms + int(e.cutoff)
        else:
            cutoff_ms = int(e.cutoff)

        if t_ms < offset_ms:
            return "before_offset"
        elif t_ms < overlap_ms:
            return "offset_to_overlap"
        elif t_ms < preutter_ms:
            return "overlap_to_preutter"
        elif t_ms < consonant_ms:
            return "preutter_to_consonant"
        elif t_ms < cutoff_ms:
            return "consonant_to_cutoff"
        else:
            return "after_cutoff"

    def get_sector_times_ms(self, sector: str) -> tuple[int, int]:
        if self._current_entry is None:
            return 0, 0

        e = self._current_entry
        dur_ms = self.duration_ms()

        offset_ms = int(e.offset)
        overlap_ms = offset_ms + int(e.overlap)
        preutter_ms = offset_ms + int(e.preutter)
        consonant_ms = offset_ms + int(e.consonant)

        if e.cutoff < 0:
            cutoff_ms = dur_ms + int(e.cutoff)
        else:
            cutoff_ms = int(e.cutoff)

        sectors = {
            "before_offset": (0, offset_ms),
            "offset_to_overlap": (offset_ms, overlap_ms),
            "overlap_to_preutter": (overlap_ms, preutter_ms),
            "preutter_to_consonant": (preutter_ms, consonant_ms),
            "consonant_to_cutoff": (consonant_ms, cutoff_ms),
            "after_cutoff": (cutoff_ms, dur_ms),
        }

        return sectors.get(sector, (0, dur_ms))

    def last_mouse_ms(self):
        t = float(max(0.0, self._last_mouse_t))

        if self._wave_times is not None and self._wave_data is not None and self._snap_enabled:
            times = self._wave_times
            data = self._wave_data

            if len(times) > 0:
                if self._snap_mode == "peaks":
                    if GPU_BACKEND_AVAILABLE and gpu_enabled():
                        from backend_gpu import get_gpu_backend
                        backend = get_gpu_backend()
                        idx = int(np.searchsorted(times, t))
                        t = backend.find_peaks(data, times, idx, 100)
                    else:
                        idx = int(np.searchsorted(times, t))
                        left = max(0, idx - 100)
                        right = min(len(times), idx + 100)
                        if right > left:
                            window = np.abs(data[left:right])
                            rel = int(np.argmax(window))
                            t = float(times[left + rel])

                elif self._snap_mode == "zero_crossing":
                    if GPU_BACKEND_AVAILABLE and gpu_enabled():
                        from backend_gpu import get_gpu_backend
                        backend = get_gpu_backend()
                        t = backend.find_zero_crossings(data, times, t, 50)
                    else:
                        idx = int(np.searchsorted(times, t))
                        if idx > 0 and idx < len(data):
                            if (data[idx - 1] * data[idx]) < 0:
                                t_left = times[idx - 1]
                                t_right = times[idx]
                                if abs(t - t_left) < abs(t - t_right):
                                    t = float(t_left)
                                else:
                                    t = float(t_right)

        if self._audio_dur > 0:
            t = max(0.0, min(t, self._audio_dur))
        return int(round(t * 1000.0))

    def set_marker_at_mouse(self, name):
        if self._current_entry is None or self._current_row is None:
            return False

        if self._wave_times is None or self._wave_data is None:
            snap_to_use = False
            wave_times_to_use = None
            wave_data_to_use = None
        else:
            snap_to_use = self._snap_enabled
            wave_times_to_use = self._wave_times
            wave_data_to_use = self._wave_data

        success = self._marker_manager.set_marker_at_mouse(
            name, self._last_mouse_t, snap_to_use, self._snap_mode,
            wave_times_to_use, wave_data_to_use,
            srp_enabled=self._srp_enabled
        )
        if success:
            self._current_entry = self._marker_manager._current_entry
            self._minimap_widget.update_minimap()
        return success

    def _cache_key_for(self, path):
        try:
            st = path.stat()
            return f"{path.resolve()}::{st.st_size}::{st.st_mtime_ns}"
        except OSError:
            return str(path.resolve())

    def _add_to_cache(self, key, value):
        self._cache[key] = value
        self._cache.move_to_end(key)

        while len(self._cache) > self._cache_max_size:
            self._cache.popitem(last=False)

    def _load_waveform_sync(self, path: Path):
        key = self._cache_key_for(path)

        if key in self._cache:
            self._cache.move_to_end(key)
            return self._cache[key]

        cache_file = self._cache_dir / (str(abs(hash(key))) + ".npz")
        if cache_file.is_file():
            try:
                data = np.load(cache_file)
                times = data["t"]
                values = data["y"]
                self._add_to_cache(key, (times, values))
                return times, values
            except Exception:
                pass

        times, values = load_waveform_sync(path, self._max_points)

        if times.size > 0:
            try:
                np.savez_compressed(cache_file, t=times, y=values)
            except Exception:
                pass

        self._add_to_cache(key, (times, values))
        return times, values

    def _load_full_waveform(self, path: Path) -> tuple[np.ndarray, int]:
        """Carrega a waveform completa para o espectrograma"""
        try:
            import wave
            with wave.open(str(path), 'rb') as wf:
                sample_rate = wf.getframerate()
                n_channels = wf.getnchannels()
                sampwidth = wf.getsampwidth()
                n_frames = wf.getnframes()
                raw = wf.readframes(n_frames)

            if sampwidth == 1:
                data = np.frombuffer(raw, dtype=np.uint8).astype(np.float32)
                data = (data - 128.0) / 128.0
            elif sampwidth == 2:
                data = np.frombuffer(raw, dtype=np.int16).astype(np.float32)
                data /= 32768.0
            else:
                data = np.frombuffer(raw, dtype=np.int16).astype(np.float32)
                max_val = np.max(np.abs(data))
                if max_val > 0:
                    data /= max_val

            if n_channels > 1:
                data = data.reshape(-1, n_channels)[:, 0]

            return data, sample_rate
        except Exception:
            return np.array([]), 44100

    def show_waveform(
            self,
            wav_path,
            entry,
            row_idx,
            reset_zoom=False,
            align_to_offset=True,
    ):
        self._current_entry = entry
        self._current_row = row_idx
        self._last_wav_path = Path(wav_path)

        times, data = self._load_waveform_sync(self._last_wav_path)
        self._wave_times = times
        self._wave_data = data

        if times.size == 0:
            self.clear()
            return

        self._audio_dur = float(times[-1])
        self._plot.set_curve_data(times, data)
        self._update_base_ylim()

        # Carrega dados completos para espectrograma se necessário
        if self._show_spectrogram:
            full_data, sample_rate = self._load_full_waveform(self._last_wav_path)
            self._wave_data_full = full_data
            self._sample_rate = sample_rate
            self._spectrogram_widget.set_audio_data(full_data, sample_rate, self._audio_dur)

        if reset_zoom:
            self._vert_zoom = 1.0
            self._horiz_zoom_state = None
        elif self._persistent_zoom and self._last_view_range is not None:
            # Mantém o zoom anterior
            self._horiz_zoom_state = self._last_view_range
        else:
            current_range = self._current_x_range()
            if self._audio_dur > 0:
                self._horiz_zoom_state = current_range

        self._apply_view_for_current_mode(align_to_offset=align_to_offset)

        self._marker_manager.set_current_entry(entry, row_idx)
        self._marker_manager.set_audio_duration(self.duration_ms())
        self._marker_manager.update_markers_from_entry(audio_duration_s=self._audio_dur)

        self._label.setText(f"{wav_path.name} – alias: {entry.alias}")
        self._minimap_widget.update_minimap()

        self.setFocus()

    def _update_spectrogram(self):
        """Atualiza o espectrograma com a região visível atual"""
        if not self._show_spectrogram or self._wave_data_full is None:
            return

        x1, x2 = self._current_x_range()
        self._spectrogram_widget.set_visible_region(x1, x2)

    def keyPressEvent(self, event):
        key = event.key()

        if key in self.MARKER_KEYS:
            marker_name = self.MARKER_KEYS[key]
            if self.set_marker_at_mouse(marker_name):
                event.accept()
                return

        if key == Qt.Key_Space:
            if event.modifiers() & Qt.ShiftModifier:
                pass
            else:
                start_ms, end_ms = self.get_segment_times_ms()
                self.playSegmentRequested.emit(float(start_ms), float(end_ms))
                event.accept()
                return

        if key == Qt.Key_Y:
            sector = self.get_mouse_sector()
            start_ms, end_ms = self.get_sector_times_ms(sector)
            self.playSegmentRequested.emit(float(start_ms), float(end_ms))
            event.accept()
            return

        if key == Qt.Key_Up:
            self.aliasStepRequested.emit(-1)
            event.accept()
            return

        if key == Qt.Key_Down:
            self.aliasStepRequested.emit(1)
            event.accept()
            return

        if key == Qt.Key_Left:
            self._pan_horizontal(120)
            event.accept()
            return

        if key == Qt.Key_Right:
            self._pan_horizontal(-120)
            event.accept()
            return

        if key == Qt.Key_Plus or key == Qt.Key_Equal:
            self.zoom_in()
            event.accept()
            return

        if key == Qt.Key_Minus:
            self.zoom_out()
            event.accept()
            return

        if key == Qt.Key_0:
            self.reset_zoom()
            event.accept()
            return

        if self._key_handler is not None and self._key_handler(event):
            return

        super().keyPressEvent(event)

    def _on_mouse_moved(self, pos):
        if self._wave_times is None:
            return
        vb = self._plot.getViewBox()
        if vb is None:
            return
        p = vb.mapSceneToView(pos)
        self._last_mouse_t = float(max(0.0, p.x()))

        self._plot.set_cursor_position(self._last_mouse_t)

    def _on_mouse_clicked(self, ev):
        self._plot.setFocus()
        self.setFocus()

    def _on_wheel(self, ev):
        delta = ev.angleDelta().y()
        mods = ev.modifiers()

        if delta == 0:
            ev.ignore()
            return

        if mods == Qt.NoModifier:
            if delta > 0:
                self.aliasStepRequested.emit(-1)
            elif delta < 0:
                self.aliasStepRequested.emit(1)
            ev.accept()
            return

        if self._audio_dur <= 0:
            ev.ignore()
            return

        if mods & Qt.ControlModifier:
            factor = 0.8 if delta > 0 else 1.25
            self._apply_horizontal_zoom(factor)
            ev.accept()
            return

        if mods & Qt.ShiftModifier:
            self._pan_horizontal(delta)
            ev.accept()
            return

        if mods & Qt.AltModifier:
            if delta > 0:
                self._vert_zoom *= 0.8
            else:
                self._vert_zoom *= 1.25
            self._update_y_range()
            ev.accept()
            return

        self._orig_wheel_event(ev)

    def _apply_horizontal_zoom(self, factor):
        if self._audio_dur <= 0:
            return
        x1, x2 = self._current_x_range()
        center = (x1 + x2) / 2.0
        span = (x2 - x1) * factor
        min_span = self._audio_dur / 200.0
        if span < min_span:
            span = min_span
        new_x1 = max(0.0, center - span / 2.0)
        new_x2 = min(self._audio_dur, center + span / 2.0)
        if new_x2 <= new_x1:
            new_x1, new_x2 = 0.0, self._audio_dur
        self._set_x_range(new_x1, new_x2)

    def _pan_horizontal(self, delta):
        if self._audio_dur <= 0:
            return
        x1, x2 = self._current_x_range()
        span = x2 - x1
        step = span * 0.1
        if delta > 0:
            x1 -= step
            x2 -= step
        else:
            x1 += step
            x2 += step
        x1 = max(0.0, x1)
        x2 = min(self._audio_dur, x2)
        if x2 <= x1:
            x1, x2 = 0.0, self._audio_dur
        self._set_x_range(x1, x2)

    def _current_x_range(self):
        if self._last_view_range is not None:
            return self._last_view_range
        vr = self._plot.get_view_range()
        if vr and len(vr[0]) == 2:
            return float(vr[0][0]), float(vr[0][1])
        return 0.0, max(self._audio_dur, 0.001)

    def _set_x_range(self, x1, x2):
        self._plot.set_x_range(x1, x2, padding=0.0)
        self._last_view_range = (x1, x2)
        self._update_minimap_view_line(x1, x2)

        # Atualiza espectrograma
        if self._show_spectrogram:
            self._spectrogram_widget.set_visible_region(x1, x2)

    def _update_minimap_view_line(self, x1, x2):
        if self._minimap_widget is None or not self._show_minimap:
            return
        self._minimap_widget.set_visible_region(x1, x2)

    def zoom_in(self):
        self._apply_horizontal_zoom(0.8)

    def zoom_out(self):
        self._apply_horizontal_zoom(1.25)

    def reset_zoom(self):
        if self._audio_dur > 0:
            self._set_x_range(0.0, self._audio_dur)
        self._vert_zoom = 1.0
        self._update_y_range()

    def _apply_view_for_current_mode(self, align_to_offset=True):
        if self._audio_dur <= 0:
            return

        # Zoom persistente tem prioridade
        if self._persistent_zoom and self._horiz_zoom_state is not None:
            x1, x2 = self._horiz_zoom_state
            x1 = max(0.0, x1)
            x2 = min(self._audio_dur, x2)
            if x2 > x1:
                self._set_x_range(x1, x2)
                self._update_y_range()
                return

        if self._always_full_view:
            self._set_x_range(0.0, self._audio_dur)
        elif self._keep_user_zoom and self._last_view_range is not None:
            self._set_x_range(*self._last_view_range)
        elif align_to_offset and self._current_entry is not None:
            center = max(0.0, float(self._current_entry.offset) / 1000.0)
            half = 0.2
            x1 = max(0.0, center - half)
            x2 = min(self._audio_dur, center + half)
            if x2 <= x1:
                x1, x2 = 0.0, self._audio_dur
            self._set_x_range(x1, x2)
        else:
            if self._last_view_range is None:
                self._set_x_range(0.0, self._audio_dur)

        self._update_y_range()

        x1, x2 = self._current_x_range()
        self._update_minimap_view_line(x1, x2)

    def _update_base_ylim(self):
        if self._wave_data is None or self._wave_data.size == 0:
            self._base_ylim = 1.0
        else:
            max_amp = float(np.max(np.abs(self._wave_data)))
            if max_amp <= 0:
                max_amp = 1.0
            self._base_ylim = max_amp * 1.1
        self._update_y_range()

    def _update_y_range(self):
        base = max(self._base_ylim, 0.1)
        y_lim = base * self._vert_zoom
        self._plot.set_y_range(-y_lim, y_lim, padding=0.0)

    def update_minimap(self):
        if not self._show_minimap:
            return
        self._minimap_widget.update_minimap()
