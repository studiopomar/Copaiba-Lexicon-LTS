class AudioPlayer:
    def __init__(self):
        self._current_stream = None
        self._audio_cache: dict[str, tuple[np.ndarray, int]] = {}
        self._cache_max_size = 20

    def stop(self):
        if self._current_stream is not None:
            try:
                sd.stop()
            except:
                pass
            self._current_stream = None

    def _load_audio(self, path: Path) -> tuple[np.ndarray, int]:
        key = str(path)
        if key in self._audio_cache:
            return self._audio_cache[key]
        try:
            with wave.open(str(path), 'rb') as wf:
                framerate = wf.getframerate()
                n_channels = wf.getnchannels()
                sampwidth = wf.getsampwidth()
                n_frames = wf.getnframes()
                raw = wf.readframes(n_frames)
            if sampwidth == 1:
                data = np.frombuffer(raw, dtype=np.uint8).astype(np.float32)
                data = (data - 128.0) / 128.0
            elif sampwidth == 2:
                data = np.frombuffer(raw, dtype=np.int16).astype(np.float32)
                data /= 32768.0
            else:
                data = np.frombuffer(raw, dtype=np.int16).astype(np.float32)
                max_val = np.max(np.abs(data))
                if max_val > 0:
                    data /= max_val
            if n_channels > 1:
                data = data.reshape(-1, n_channels)
            if len(self._audio_cache) >= self._cache_max_size:
                oldest = next(iter(self._audio_cache))
                del self._audio_cache[oldest]
            self._audio_cache[key] = (data, framerate)
            return data, framerate
        except Exception as e:
            print(f"Erro ao carregar áudio: {e}")
            return np.array([]), 44100

    def play_segment(self, path: Path, start_ms: float, end_ms: float):
        self.stop()
        data, framerate = self._load_audio(path)
        if data.size == 0:
            return
        start_sample = int(start_ms * framerate / 1000.0)
        end_sample = int(end_ms * framerate / 1000.0)
        start_sample = max(0, start_sample)
        if len(data.shape) == 1:
            end_sample = min(len(data), end_sample)
            segment = data[start_sample:end_sample]
        else:
            end_sample = min(len(data), end_sample)
            segment = data[start_sample:end_sample]
        if segment.size == 0:
            return
        try:
            sd.play(segment, framerate)
            self._current_stream = True
        except Exception as e:
            print(f"Erro ao reproduzir: {e}")

    def play_full(self, path: Path):
        self.stop()
        data, framerate = self._load_audio(path)
        if data.size == 0:
            return
        try:
            sd.play(data, framerate)
            self._current_stream = True
        except Exception as e:
            print(f"Erro ao reproduzir: {e}")


class PresetConfigWidget(QWidget):
    def __init__(self, parent=None):
        super().__init__(parent)
        self._custom_presets = {}
        self._presets_active = True
        self._init_ui()
        self._load_from_defaults()

    def _init_ui(self):
        layout = QVBoxLayout(self)
        layout.setContentsMargins(5, 5, 5, 5)
        layout.setSpacing(10)

        self._active_checkbox = QCheckBox("Presets Ativos")
        self._active_checkbox.setChecked(True)
        self._active_checkbox.setToolTip("Quando desativado, os botões de preset não terão efeito")
        self._active_checkbox.toggled.connect(self._on_active_toggled)
        layout.addWidget(self._active_checkbox)

        self._preset_widgets = {}
        preset_configs = [("cv", "CV"), ("vcv", "VCV"), ("vv", "VV"), ("vc", "VC"), ("minus_v", "-V")]

        for key, label in preset_configs:
            group = QGroupBox(f"Preset {label}")
            group_layout = QGridLayout(group)
            group_layout.setSpacing(5)

            spin_overlap = QSpinBox()
            spin_overlap.setRange(-1000, 1000)
            spin_overlap.setSuffix(" ms")

            spin_preutter = QSpinBox()
            spin_preutter.setRange(0, 2000)
            spin_preutter.setSuffix(" ms")

            spin_consonant = QSpinBox()
            spin_consonant.setRange(0, 2000)
            spin_consonant.setSuffix(" ms")

            spin_cutoff = QSpinBox()
            spin_cutoff.setRange(-2000, 0)
            spin_cutoff.setSuffix(" ms")

            group_layout.addWidget(QLabel("Overlap:"), 0, 0)
            group_layout.addWidget(spin_overlap, 0, 1)
            group_layout.addWidget(QLabel("Preutter:"), 0, 2)
            group_layout.addWidget(spin_preutter, 0, 3)
            group_layout.addWidget(QLabel("Consonant:"), 1, 0)
            group_layout.addWidget(spin_consonant, 1, 1)
            group_layout.addWidget(QLabel("Cutoff:"), 1, 2)
            group_layout.addWidget(spin_cutoff, 1, 3)

            btn_reset = QPushButton("Reset")
            btn_reset.clicked.connect(lambda checked, k=key: self._reset_preset(k))
            group_layout.addWidget(btn_reset, 2, 0, 1, 4)

            self._preset_widgets[key] = {
                "overlap": spin_overlap, "preutter": spin_preutter,
                "consonant": spin_consonant, "cutoff": spin_cutoff, "group": group,
            }
            layout.addWidget(group)
        layout.addStretch()

    def _on_active_toggled(self, checked: bool):
        self._presets_active = checked
        for key, widgets in self._preset_widgets.items():
            widgets["group"].setEnabled(checked)

    def is_active(self) -> bool:
        return self._presets_active

    def set_active(self, active: bool):
        self._presets_active = active
        self._active_checkbox.setChecked(active)

    def _load_from_defaults(self):
        for key, preset in PRESETS.items():
            if key in self._preset_widgets:
                widgets = self._preset_widgets[key]
                widgets["overlap"].setValue(preset.overlap)
                widgets["preutter"].setValue(preset.preutter)
                widgets["consonant"].setValue(preset.consonant)
                widgets["cutoff"].setValue(preset.cutoff)

    def _reset_preset(self, key: str):
        if key in PRESETS and key in self._preset_widgets:
            preset = PRESETS[key]
            widgets = self._preset_widgets[key]
            widgets["overlap"].setValue(preset.overlap)
            widgets["preutter"].setValue(preset.preutter)
            widgets["consonant"].setValue(preset.consonant)
            widgets["cutoff"].setValue(preset.cutoff)

    def get_preset(self, key: str) -> Preset:
        if key in self._preset_widgets:
            widgets = self._preset_widgets[key]
            return Preset(
                name=f"Custom {key.upper()}",
                overlap=widgets["overlap"].value(),
                preutter=widgets["preutter"].value(),
                consonant=widgets["consonant"].value(),
                cutoff=widgets["cutoff"].value(),
            )
        return PRESETS.get(key, PRESETS["cv"])

    def get_all_presets(self) -> dict:
        result = {}
        for key in self._preset_widgets:
            widgets = self._preset_widgets[key]
            result[key] = {
                "overlap": widgets["overlap"].value(),
                "preutter": widgets["preutter"].value(),
                "consonant": widgets["consonant"].value(),
                "cutoff": widgets["cutoff"].value(),
            }
        result["_active"] = self._presets_active
        return result

    def set_all_presets(self, presets: dict):
        for key, values in presets.items():
            if key == "_active":
                self.set_active(values)
                continue
            if key in self._preset_widgets:
                widgets = self._preset_widgets[key]
                widgets["overlap"].setValue(values.get("overlap", 0))
                widgets["preutter"].setValue(values.get("preutter", 0))
                widgets["consonant"].setValue(values.get("consonant", 0))
                widgets["cutoff"].setValue(values.get("cutoff", 0))

    def _table_mouse_press_event(self, event):
        if event.button() == Qt.LeftButton:
            pos = event.position().toPoint()
            item = self.table.itemAt(pos)
            if item:
                row = item.row()
                if event.modifiers() & Qt.ShiftModifier:
                    if self._last_shift_click_row is not None:
                        start_row = min(self._last_shift_click_row, row)
                        end_row = max(self._last_shift_click_row, row)
                        self.table.clearSelection()
                        for r in range(start_row, end_row + 1):
                            for c in range(self.table.columnCount()):
                                if self.table.item(r, c):
                                    self.table.item(r, c).setSelected(True)
                    else:
                        self._last_shift_click_row = row
                        QTableWidget.mousePressEvent(self.table, event)
                else:
                    self._last_shift_click_row = row
                    QTableWidget.mousePressEvent(self.table, event)
            else:
                QTableWidget.mousePressEvent(self.table, event)
        else:
            QTableWidget.mousePressEvent(self.table, event)

    def _table_mouse_release_event(self, event):
        QTableWidget.mouseReleaseEvent(self.table, event)
        self._update_line_indicator()

    def keyPressEvent(self, event):
        if event.matches(QKeySequence.StandardKey.Copy):
            self._copy_selection()
            event.accept()
            return
        elif event.matches(QKeySequence.StandardKey.Paste):
            self._paste_selection()
            event.accept()
            return
        elif event.key() == Qt.Key.Key_Space:
            if event.modifiers() & Qt.ShiftModifier:
                self._play_full_audio()
            else:
                self._play_segment()
            event.accept()
            return
        super().keyPressEvent(event)

    def _copy_selection(self):
        selection = self.table.selectedItems()
        if not selection:
            return
        items_by_position = {}
        cols_used = set()
        for item in selection:
            row, col = item.row(), item.column()
            if row not in items_by_position:
                items_by_position[row] = {}
            items_by_position[row][col] = item.text()
            cols_used.add(col)
        if items_by_position:
            min_row = min(items_by_position.keys())
            max_row = max(items_by_position.keys())
            min_col = min(cols_used)
            max_col = max(cols_used)
            self._clipboard_data = []
            self._clipboard_cols = list(range(min_col, max_col + 1))
            for r in range(min_row, max_row + 1):
                row_data = []
                for c in range(min_col, max_col + 1):
                    if r in items_by_position and c in items_by_position[r]:
                        row_data.append(items_by_position[r][c])
                    else:
                        row_data.append("")
                self._clipboard_data.append(row_data)
            self.statusBar().showMessage(f"Copiadas {len(selection)} células", 2000)

    def _paste_selection(self):
        if not self._clipboard_data:
            self.statusBar().showMessage("Clipboard vazio", 2000)
            return
        selection = self.table.selectedItems()
        if not selection:
            return
        selected_rows = sorted(set(item.row() for item in selection))
        selected_cols = sorted(set(item.column() for item in selection))
        edits: list[CellEdit] = []
        self._updating_from_code = True
        try:
            if len(self._clipboard_data) == 1 and len(self._clipboard_data[0]) == 1:
                value = self._clipboard_data[0][0]
                for item in selection:
                    row, col = item.row(), item.column()
                    if col == self.COL_FAV:
                        continue
                    old_val = item.text()
                    new_val = value
                    if col in [self.COL_PREUTTER, self.COL_CONSONANT]:
                        try:
                            num_val = int(round(float(new_val)))
                            if num_val < 0:
                                new_val = "0"
                        except ValueError:
                            continue
                    if old_val != new_val:
                        edits.append(CellEdit(row, col, old_val, new_val))
                        item.setText(new_val)
                        item.setData(Qt.ItemDataRole.UserRole, new_val)
            else:
                clipboard_width = len(self._clipboard_data[0]) if self._clipboard_data else 0
                clipboard_height = len(self._clipboard_data)
                for idx, target_row in enumerate(selected_rows):
                    clipboard_row_idx = idx % clipboard_height
                    clipboard_row = self._clipboard_data[clipboard_row_idx]
                    for col_idx, target_col in enumerate(selected_cols):
                        if target_col == self.COL_FAV:
                            continue
                        clipboard_col_idx = col_idx % clipboard_width
                        value = clipboard_row[clipboard_col_idx] if clipboard_col_idx < len(clipboard_row) else ""
                        if not value:
                            continue
                        item = self.table.item(target_row, target_col)
                        if item is None:
                            continue
                        old_val = item.text()
                        new_val = value
                        if target_col in [self.COL_PREUTTER, self.COL_CONSONANT]:
                            try:
                                num_val = int(round(float(new_val)))
                                if num_val < 0:
                                    new_val = "0"
                            except ValueError:
                                continue
                        if old_val != new_val:
                            edits.append(CellEdit(target_row, target_col, old_val, new_val))
                            item.setText(new_val)
                            item.setData(Qt.ItemDataRole.UserRole, new_val)
        finally:
            self._updating_from_code = False
        if edits:
            self._push_undo(edits)
            self.statusBar().showMessage(f"Colado em {len(edits)} células", 2000)
            self._load_waveform_for_current_row()

    def _update_line_indicator(self):
        current_row = self.table.currentRow()
        total_rows = self.table.rowCount()
        if current_row >= 0:
            self._line_indicator_label.setText(f"Linha: {current_row + 1}/{total_rows}")
        else:
            self._line_indicator_label.setText("Linha: -/-")

    def _get_last_saved_string(self) -> str:
        if self._last_saved_time is None:
            return "Nunca"
        now = datetime.now()
        diff = now - self._last_saved_time
        if diff.total_seconds() < 60:
            return "Agora há pouco"
        elif diff.total_seconds() < 3600:
            minutes = int(diff.total_seconds() / 60)
            return f"Há {minutes} minuto{'s' if minutes > 1 else ''}"
        elif diff.total_seconds() < 86400:
            hours = int(diff.total_seconds() / 3600)
            return f"Há {hours} hora{'s' if hours > 1 else ''}"
        else:
            return self._last_saved_time.strftime("%d/%m/%Y às %H:%M")

    def _create_backup(self) -> bool:
        if self._current_path is None or not self._current_path.exists():
            return False
        try:
            timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
            backup_name = f"{self._current_path.stem}_backup_{timestamp}{self._current_path.suffix}"
            backup_path = self._current_path.parent / backup_name
            shutil.copy2(self._current_path, backup_path)
            backup_pattern = f"{self._current_path.stem}_backup_*{self._current_path.suffix}"
            backups = list(self._current_path.parent.glob(backup_pattern))
            backups.sort(key=lambda x: x.stat().st_mtime, reverse=True)
            for old_backup in backups[10:]:
                try:
                    old_backup.unlink()
                except OSError:
                    pass
            return True
        except Exception as e:
            QMessageBox.warning(self, "Erro no Backup", f"Não foi possível criar backup:\n{e}")
            return False

    def open_voicebank_folder(self):
        folder = QFileDialog.getExistingDirectory(self, "Selecionar pasta do voicebank")
        if not folder:
            return
        folder_path = Path(folder)
        oto_path = folder_path / "oto.ini"
        if oto_path.exists():
            self._voicebank_dir = folder_path
            self._load_oto(oto_path)
        else:
            QMessageBox.warning(self, "Erro", "Nenhum oto.ini encontrado nesta pasta.")

    def open_oto(self):
        path, _ = QFileDialog.getOpenFileName(
            self, "Abrir oto.ini", "", "OTO Files (oto.ini);;All Files (*)"
        )
        if path:
            self._load_oto(Path(path))

    def _load_oto(self, path: Path):
        try:
            self._oto.load(str(path), encoding=self._encoding)
            self._current_path = path
            self._voicebank_dir = path.parent
            self._dirty = False
            self._undo_stack.clear()
            self._redo_stack.clear()
            self._load_oto_to_table()
            self._update_title()
            self._update_progress()
            self.statusBar().showMessage(f"Carregado: {path.name}", 3000)
        except Exception as e:
            QMessageBox.critical(self, "Erro ao carregar", str(e))

    def _load_oto_to_table(self):
        self._updating_from_code = True
        self.table.setRowCount(0)
        self.table.setRowCount(len(self._oto.entries))

        for row, entry in enumerate(self._oto.entries):
            # Coluna de marcação
            fav_item = QTableWidgetItem()
            fav_item.setFlags(Qt.ItemIsSelectable | Qt.ItemIsEnabled | Qt.ItemIsUserCheckable)
            key = f"{entry.filename}|{entry.alias}"
            if key in self._completed_aliases:
                fav_item.setCheckState(Qt.Checked)
            else:
                fav_item.setCheckState(Qt.Unchecked)
            self.table.setItem(row, self.COL_FAV, fav_item)

            # Filename
            fn_item = QTableWidgetItem(entry.filename)
            fn_item.setFlags(Qt.ItemIsSelectable | Qt.ItemIsEnabled)
            self.table.setItem(row, self.COL_FILENAME, fn_item)

            # Alias
            alias_item = QTableWidgetItem(entry.alias)
            self.table.setItem(row, self.COL_ALIAS, alias_item)

            # Valores numéricos
            self.table.setItem(row, self.COL_OFFSET, QTableWidgetItem(str(entry.offset)))
            self.table.setItem(row, self.COL_OVERLAP, QTableWidgetItem(str(entry.overlap)))
            self.table.setItem(row, self.COL_PREUTTER, QTableWidgetItem(str(entry.preutter)))
            self.table.setItem(row, self.COL_CONSONANT, QTableWidgetItem(str(entry.consonant)))
            self.table.setItem(row, self.COL_CUTOFF, QTableWidgetItem(str(entry.cutoff)))

        self._updating_from_code = False
        self._update_undo_redo_actions()

    def save_oto(self):
        if self._current_path is None:
            self.save_oto_as()
            return
        try:
            self._sync_table_to_oto()
            self._oto.save(str(self._current_path), encoding=self._encoding)
            self._dirty = False
            self._last_saved_time = datetime.now()
            self._update_title()
            self.statusBar().showMessage("Salvo com sucesso!", 3000)
        except Exception as e:
            QMessageBox.critical(self, "Erro ao salvar", str(e))

    def save_oto_as(self):
        path, _ = QFileDialog.getSaveFileName(
            self, "Salvar oto.ini como", "", "OTO Files (*.ini);;All Files (*)"
        )
        if path:
            self._current_path = Path(path)
            self.save_oto()

    def _sync_table_to_oto(self):
        """Sincroniza os dados da tabela para o objeto OtoFile"""
        self._oto.entries.clear()
        for row in range(self.table.rowCount()):
            filename = self.table.item(row, self.COL_FILENAME).text()
            alias = self.table.item(row, self.COL_ALIAS).text()
            offset = int(self.table.item(row, self.COL_OFFSET).text() or "0")
            overlap = int(self.table.item(row, self.COL_OVERLAP).text() or "0")
            preutter = int(self.table.item(row, self.COL_PREUTTER).text() or "0")
            consonant = int(self.table.item(row, self.COL_CONSONANT).text() or "0")
            cutoff = int(self.table.item(row, self.COL_CUTOFF).text() or "0")

            entry = OtoEntry(
                filename=filename,
                alias=alias,
                offset=offset,
                consonant=consonant,
                cutoff=cutoff,
                preutter=preutter,
                overlap=overlap,
            )
            self._oto.entries.append(entry)

    def reload_oto(self):
        if self._current_path and self._current_path.exists():
            if self._dirty:
                reply = QMessageBox.question(
                    self, "Recarregar",
                    "Existem alterações não salvas. Deseja recarregar mesmo assim?",
                    QMessageBox.Yes | QMessageBox.No
                )
                if reply != QMessageBox.Yes:
                    return
            self._load_oto(self._current_path)

    def reveal_voicebank(self):
        if self._voicebank_dir and self._voicebank_dir.exists():
            import subprocess
            import platform
            if platform.system() == "Windows":
                subprocess.run(["explorer", str(self._voicebank_dir)])
            elif platform.system() == "Darwin":
                subprocess.run(["open", str(self._voicebank_dir)])
            else:
                subprocess.run(["xdg-open", str(self._voicebank_dir)])

    def open_project(self):
        path, _ = QFileDialog.getOpenFileName(
            self, "Abrir projeto", "", "Copaiba Project (*.copaiba);;All Files (*)"
        )
        if not path:
            return
        try:
            with open(path, "r", encoding="utf-8") as f:
                data = json.load(f)

                QMessageBox.warning(self, "Erro", f"O arquivo oto.ini não foi encontrado:\n{oto_path}")

    def _on_item_changed(self, item: QTableWidgetItem):
        if self._updating_from_code or self._in_undo_redo:
            return

        row, col = item.row(), item.column()

        # Tratamento especial para checkbox de conclusão
        if col == self.COL_FAV:
            self._on_complete_toggled(row, item.checkState() == Qt.Checked)
            return

        old_val = item.data(Qt.ItemDataRole.UserRole)
        new_val = item.text()

        if old_val is None:
            old_val = ""

        if old_val != new_val:
            # Validação para colunas numéricas
            if col in [self.COL_OFFSET, self.COL_OVERLAP, self.COL_PREUTTER, self.COL_CONSONANT, self.COL_CUTOFF]:
                try:
                    int(new_val)
                except ValueError:
                    self._updating_from_code = True
                    item.setText(str(old_val))
                    self._updating_from_code = False
                    return

            edit = CellEdit(row, col, str(old_val), new_val)
            self._push_undo([edit])
            item.setData(Qt.ItemDataRole.UserRole, new_val)
            self._dirty = True
            self._update_title()
            self._load_waveform_for_current_row()

    def _on_complete_toggled(self, row: int, completed: bool):
        if row >= len(self._oto.entries):
            return
        entry = self._oto.entries[row]
        key = f"{entry.filename}|{entry.alias}"
        if completed:
            self._completed_aliases.add(key)
        else:
            self._completed_aliases.discard(key)
        self._update_progress()

    def _on_selection_changed(self):
        self._load_waveform_for_current_row()

    def _on_current_cell_changed(self, current_row, current_col, previous_row, previous_col):
        if current_row != previous_row and current_row >= 0:
            self._load_waveform_for_current_row()
        self._update_line_indicator()

    def _load_waveform_for_current_row(self):
        row = self.table.currentRow()
        if row < 0 or row >= len(self._oto.entries):
            self.waveform.clear()
            return

        entry = self._oto.entries[row]

        # Atualiza entry com valores da tabela
        entry = OtoEntry(
            filename=self.table.item(row, self.COL_FILENAME).text(),
            alias=self.table.item(row, self.COL_ALIAS).text(),
            offset=int(self.table.item(row, self.COL_OFFSET).text() or "0"),
            consonant=int(self.table.item(row, self.COL_CONSONANT).text() or "0"),
            cutoff=int(self.table.item(row, self.COL_CUTOFF).text() or "0"),
            preutter=int(self.table.item(row, self.COL_PREUTTER).text() or "0"),
            overlap=int(self.table.item(row, self.COL_OVERLAP).text() or "0"),
        )

        if self._voicebank_dir:
            wav_path = self._voicebank_dir / entry.filename
            if wav_path.exists():
                reset_zoom = not self._keep_zoom_on_alias_change
                self.waveform.show_waveform(wav_path, entry, row, reset_zoom=reset_zoom)
            else:
                self.waveform.clear()
                self.statusBar().showMessage(f"Arquivo não encontrado: {entry.filename}", 3000)

    def _entry_edited_from_waveform(self, row: int, entry: OtoEntry):
        """Callback quando os marcadores são editados no waveform"""
        if row < 0 or row >= self.table.rowCount():
            return

        edits = []
        self._updating_from_code = True
        try:
            col_map = {
                self.COL_OFFSET: str(int(entry.offset)),
                self.COL_OVERLAP: str(int(entry.overlap)),
                self.COL_PREUTTER: str(int(entry.preutter)),
                self.COL_CONSONANT: str(int(entry.consonant)),
                self.COL_CUTOFF: str(int(entry.cutoff)),
            }

            for col, new_val in col_map.items():
                item = self.table.item(row, col)
                if item:
                    old_val = item.text()
                    if old_val != new_val:
                        edits.append(CellEdit(row, col, old_val, new_val))
                        item.setText(new_val)
                        item.setData(Qt.ItemDataRole.UserRole, new_val)

        finally:
            self._updating_from_code = False

        if edits:
            self._push_undo(edits)
            self._dirty = True
            self._update_title()

    def apply_preset(self, preset_key: str):
        """Aplica preset aos aliases selecionados"""
        if not self.preset_config.is_active():
            self.statusBar().showMessage("Presets estão desativados nas configurações", 2000)
            return

        selected_rows = sorted(set(item.row() for item in self.table.selectedItems()))
        if not selected_rows:
            self.statusBar().showMessage("Nenhum alias selecionado", 2000)
            return

        preset = self.preset_config.get_preset(preset_key)

        edits = []
        self._updating_from_code = True
        try:
            for row in selected_rows:
                # Overlap
                item = self.table.item(row, self.COL_OVERLAP)
                if item:
                    old_val = item.text()
                    new_val = str(preset.overlap)
                    if old_val != new_val:
                        edits.append(CellEdit(row, self.COL_OVERLAP, old_val, new_val))
                        item.setText(new_val)
                        item.setData(Qt.ItemDataRole.UserRole, new_val)

                # Preutter
                item = self.table.item(row, self.COL_PREUTTER)
                if item:
                    old_val = item.text()
                    new_val = str(preset.preutter)
                    if old_val != new_val:
                        edits.append(CellEdit(row, self.COL_PREUTTER, old_val, new_val))
                        item.setText(new_val)
                        item.setData(Qt.ItemDataRole.UserRole, new_val)

                # Consonant
                item = self.table.item(row, self.COL_CONSONANT)
                if item:
                    old_val = item.text()
                    new_val = str(preset.consonant)
                    if old_val != new_val:
                        edits.append(CellEdit(row, self.COL_CONSONANT, old_val, new_val))
                        item.setText(new_val)
                        item.setData(Qt.ItemDataRole.UserRole, new_val)

                # Cutoff
                item = self.table.item(row, self.COL_CUTOFF)
                if item:
                    old_val = item.text()
                    new_val = str(preset.cutoff)
                    if old_val != new_val:
                        edits.append(CellEdit(row, self.COL_CUTOFF, old_val, new_val))
                        item.setText(new_val)
                        item.setData(Qt.ItemDataRole.UserRole, new_val)

        finally:
            self._updating_from_code = False

        if edits:
            self._push_undo(edits)
            self._dirty = True
            self._update_title()
            self._load_waveform_for_current_row()
            self.statusBar().showMessage(f"Preset {preset_key.upper()} aplicado a {len(selected_rows)} alias(es)", 2000)

    # =========================================================================
    # Undo/Redo
    # =========================================================================

    def _push_undo(self, edit: Union[list[CellEdit], RowEdit]):
        """Adiciona edição ao stack de undo"""
        self._undo_stack.append(edit)
        self._redo_stack.clear()
        self._update_undo_redo_actions()

    def undo(self):
        """Desfaz última edição"""
        if not self._undo_stack:
            return

        edit = self._undo_stack.pop()
        self._in_undo_redo = True
        self._updating_from_code = True

        try:
            if isinstance(edit, list):
                for cell_edit in reversed(edit):
                    item = self.table.item(cell_edit.row, cell_edit.col)
                    if item:
                        item.setText(cell_edit.old)
                        item.setData(Qt.ItemDataRole.UserRole, cell_edit.old)
                self._redo_stack.append(edit)

            elif isinstance(edit, RowEdit):
                if edit.is_insert:
                    self.table.removeRow(edit.row)
                else:
                    self.table.insertRow(edit.row)
                    for col, value in enumerate(edit.data):
                        self.table.setItem(edit.row, col, QTableWidgetItem(value))
                self._redo_stack.append(edit)

        finally:
            self._in_undo_redo = False
            self._updating_from_code = False

        self._dirty = True
        self._update_title()
        self._update_undo_redo_actions()
        self._load_waveform_for_current_row()

    def redo(self):
        """Refaz última edição desfeita"""
        if not self._redo_stack:
            return

        edit = self._redo_stack.pop()
        self._in_undo_redo = True
        self._updating_from_code = True

        try:
            if isinstance(edit, list):
                for cell_edit in edit:
                    item = self.table.item(cell_edit.row, cell_edit.col)
                    if item:
                        item.setText(cell_edit.new)
                        item.setData(Qt.ItemDataRole.UserRole, cell_edit.new)
                self._undo_stack.append(edit)

            elif isinstance(edit, RowEdit):
                if edit.is_insert:
                    self.table.insertRow(edit.row)
                    for col, value in enumerate(edit.data):
                        self.table.setItem(edit.row, col, QTableWidgetItem(value))
                else:
                    self.table.removeRow(edit.row)
                self._undo_stack.append(edit)

        finally:
            self._in_undo_redo = False
            self._updating_from_code = False

        self._dirty = True
        self._update_title()
        self._update_undo_redo_actions()
        self._load_waveform_for_current_row()

    def _update_undo_redo_actions(self):
        """Atualiza estado dos botões de undo/redo"""
        self.act_undo.setEnabled(len(self._undo_stack) > 0)
        self.act_redo.setEnabled(len(self._redo_stack) > 0)

    def rename_alias(self):
        """Renomeia o alias atual"""
        row = self.table.currentRow()
        if row < 0:
            return

        alias_item = self.table.item(row, self.COL_ALIAS)
        if not alias_item:
            return

        old_alias = alias_item.text()
        new_alias, ok = QInputDialog.getText(
            self, "Renomear Alias", "Novo nome:", text=old_alias
        )

        if ok and new_alias and new_alias != old_alias:
            self._updating_from_code = True
            alias_item.setText(new_alias)
            alias_item.setData(Qt.ItemDataRole.UserRole, new_alias)
            self._updating_from_code = False

            edit = CellEdit(row, self.COL_ALIAS, old_alias, new_alias)
            self._push_undo([edit])
            self._dirty = True
            self._update_title()
            self._load_waveform_for_current_row()

    def delete_alias(self):
        """Deleta o alias atual"""
        row = self.table.currentRow()
        if row < 0:
            return

        reply = QMessageBox.question(
            self, "Confirmar Exclusão",
            f"Deseja realmente excluir o alias da linha {row + 1}?",
            QMessageBox.StandardButton.Yes | QMessageBox.StandardButton.No
        )

        if reply == QMessageBox.StandardButton.Yes:
            row_data = []
            for col in range(self.table.columnCount()):
                item = self.table.item(row, col)
                row_data.append(item.text() if item else "")

            self.table.removeRow(row)

            edit = RowEdit(row, row_data, is_insert=False)
            self._push_undo(edit)
            self._dirty = True
            self._update_title()
            self._load_waveform_for_current_row()

    def duplicate_alias(self):
        """Duplica o alias atual"""
        row = self.table.currentRow()
        if row < 0:
            return

        row_data = []
        for col in range(self.table.columnCount()):
            item = self.table.item(row, col)
            row_data.append(item.text() if item else "")

        new_row = row + 1
        self.table.insertRow(new_row)

        self._updating_from_code = True
        for col, value in enumerate(row_data):
            if col == self.COL_FAV:
                fav_item = QTableWidgetItem()
                fav_item.setFlags(Qt.ItemFlag.ItemIsSelectable | Qt.ItemFlag.ItemIsEnabled | Qt.ItemFlag.ItemIsUserCheckable)
                fav_item.setCheckState(Qt.CheckState.Unchecked)
                self.table.setItem(new_row, col, fav_item)
            elif col == self.COL_FILENAME:
                fn_item = QTableWidgetItem(value)
                fn_item.setFlags(Qt.ItemFlag.ItemIsSelectable | Qt.ItemFlag.ItemIsEnabled)
                self.table.setItem(new_row, col, fn_item)
            else:
                item = QTableWidgetItem(value)
                item.setData(Qt.ItemDataRole.UserRole, value)
                self.table.setItem(new_row, col, item)
        self._updating_from_code = False

        alias_item = self.table.item(new_row, self.COL_ALIAS)
        if alias_item:
            old_alias = alias_item.text()
            alias_item.setText(f"{old_alias} (cópia)")

        edit = RowEdit(new_row, row_data, is_insert=True)
        self._push_undo(edit)
        self._dirty = True
        self._update_title()
        self.table.setCurrentCell(new_row, self.COL_ALIAS)

    def _toggle_complete_current(self):
        """Marca/desmarca o alias atual como concluído"""
        row = self.table.currentRow()
        if row < 0:
            return

        fav_item = self.table.item(row, self.COL_FAV)
        if fav_item:
            if fav_item.checkState() == Qt.CheckState.Checked:
                fav_item.setCheckState(Qt.CheckState.Unchecked)
            else:
                fav_item.setCheckState(Qt.CheckState.Checked)

    def _step_alias(self, direction: int):
        """Navega entre aliases (+1 = próximo, -1 = anterior)"""
        current_row = self.table.currentRow()
        if current_row < 0:
            return

        new_row = current_row + direction
        if 0 <= new_row < self.table.rowCount():
            self.table.setCurrentCell(new_row, self.table.currentColumn())

    def _handle_waveform_key(self, event) -> bool:
        """Manipula eventos de teclado no waveform"""
        return False

    def _filter_table(self, text: str):
        """Filtra a tabela por texto"""
        for row in range(self.table.rowCount()):
            should_show = True
            if text.strip():
                alias_item = self.table.item(row, self.COL_ALIAS)
                filename_item = self.table.item(row, self.COL_FILENAME)

                alias_text = alias_item.text() if alias_item else ""
                filename_text = filename_item.text() if filename_item else ""

                should_show = (text.lower() in alias_text.lower() or
                               text.lower() in filename_text.lower())

            self.table.setRowHidden(row, not should_show)

    def toggle_minimap(self, checked: bool):
        """Mostra/oculta o minimapa"""
        self.waveform.set_show_minimap(checked)

    def _toggle_spectrogram(self, checked: bool):
        """Mostra/oculta o espectrograma"""
        self.waveform.set_show_spectrogram(checked)

    def _toggle_spectrogram_config_dock(self):
        """Mostra/oculta dock de configuração do espectrograma"""
        if self._dock_spectrogram_config.isVisible():
            self._dock_spectrogram_config.hide()
        else:
            self._dock_spectrogram_config.show()

    def _toggle_preset_dock(self):
        """Mostra/oculta dock de configuração de presets"""
        if self._dock_presets.isVisible():
            self._dock_presets.hide()
        else:
            self._dock_presets.show()

    def _toggle_gpu_acceleration(self, checked: bool):
        """Ativa/desativa aceleração GPU"""
        if not GPU_BACKEND_AVAILABLE:
            self.statusBar().showMessage("Backend GPU não disponível", 2000)
            self.act_toggle_gpu.setChecked(False)
            return

        if checked:
            if enable_gpu():
                self.statusBar().showMessage("GPU ativada com sucesso", 2000)
            else:
                self.statusBar().showMessage("Falha ao ativar GPU", 2000)
                self.act_toggle_gpu.setChecked(False)
        else:
            disable_gpu()
            self.statusBar().showMessage("GPU desativada", 2000)

        self._update_gpu_status_label()

    def _show_gpu_info(self):
        """Mostra informações sobre a GPU"""
        if not GPU_BACKEND_AVAILABLE:
            QMessageBox.information(
                self, "GPU Info",
                "Backend GPU não disponível.\n\nInstale CuPy (NVIDIA) ou PyOpenCL (AMD/Intel) para aceleração GPU."
            )
            return

        info = get_gpu_info()
        status = "✅ Ativada" if gpu_enabled() else "⚠️ Disponível (desativada)"

        msg = f"""
<b>Informações da GPU</b><br><br>
<b>Dispositivo:</b> {info}<br>
<b>Status:</b> {status}<br>
<b>Vendor:</b> {info.vendor.value.upper()}<br>
<b>Backend:</b> {info.backend}<br>
"""

        if info.vendor.value != "cpu":
            msg += f"<b>Memória:</b> {info.memory_gb:.1f} GB<br>"
            msg += f"<b>Compute:</b> {info.compute_capability}<br>"

        QMessageBox.information(self, "GPU Info", msg)

    def _toggle_auto_save(self, checked: bool):
        """Ativa/desativa salvamento automático"""
        if checked:
            if self._auto_save_timer is None:
                self._auto_save_timer = QTimer(self)
                self._auto_save_timer.timeout.connect(self._auto_save)
            self._auto_save_timer.start(self._auto_save_interval * 1000)
            self.statusBar().showMessage(f"Auto-save ativado ({self._auto_save_interval}s)", 2000)
        else:
            if self._auto_save_timer:
                self._auto_save_timer.stop()
            self.statusBar().showMessage("Auto-save desativado", 2000)

    def _auto_save(self):
        """Salva automaticamente se houver mudanças"""
        if self._dirty and self._current_path:
            self.save_oto()

    def _open_batch_edit_dialog(self):
        """Abre diálogo de edição em lote"""
        selected_rows = sorted(set(item.row() for item in self.table.selectedItems()))
        if not selected_rows:
            QMessageBox.information(self, "Edição em Lote", "Selecione pelo menos um alias.")
            return

        dialog = BatchEditDialog(self, len(selected_rows))
        if dialog.exec() == QDialog.DialogCode.Accepted:
            values = dialog.get_values()
            if not values:
                return

            edits = []
            self._updating_from_code = True
            try:
                for row in selected_rows:
                    if "offset" in values:
                        item = self.table.item(row, self.COL_OFFSET)
                        if item:
                            old_val = item.text()
                            new_val = str(values["offset"])
                            if old_val != new_val:
                                edits.append(CellEdit(row, self.COL_OFFSET, old_val, new_val))
                                item.setText(new_val)
                                item.setData(Qt.ItemDataRole.UserRole, new_val)

                    if "overlap" in values:
                        item = self.table.item(row, self.COL_OVERLAP)
                        if item:
                            old_val = item.text()
                            new_val = str(values["overlap"])
                            if old_val != new_val:
                                edits.append(CellEdit(row, self.COL_OVERLAP, old_val, new_val))
                                item.setText(new_val)
                                item.setData(Qt.ItemDataRole.UserRole, new_val)

                    if "preutter" in values:
                        item = self.table.item(row, self.COL_PREUTTER)
                        if item:
                            old_val = item.text()
                            new_val = str(values["preutter"])
                            if old_val != new_val:
                                edits.append(CellEdit(row, self.COL_PREUTTER, old_val, new_val))
                                item.setText(new_val)
                                item.setData(Qt.ItemDataRole.UserRole, new_val)

                    if "consonant" in values:
                        item = self.table.item(row, self.COL_CONSONANT)
                        if item:
                            old_val = item.text()
                            new_val = str(values["consonant"])
                            if old_val != new_val:
                                edits.append(CellEdit(row, self.COL_CONSONANT, old_val, new_val))
                                item.setText(new_val)
                                item.setData(Qt.ItemDataRole.UserRole, new_val)

                    if "cutoff" in values:
                        item = self.table.item(row, self.COL_CUTOFF)
                        if item:
                            old_val = item.text()
                            new_val = str(values["cutoff"])
                            if old_val != new_val:
                                edits.append(CellEdit(row, self.COL_CUTOFF, old_val, new_val))
                                item.setText(new_val)
                                item.setData(Qt.ItemDataRole.UserRole, new_val)

            finally:
                self._updating_from_code = False

            if edits:
                self._push_undo(edits)
                self._dirty = True
                self._update_title()
                self._load_waveform_for_current_row()
                self.statusBar().showMessage(f"Edição em lote aplicada a {len(selected_rows)} alias(es)", 2000)

    def _set_encoding(self, encoding: str):

        if encoding == "auto":
            self.act_encoding_auto.setChecked(True)
        elif encoding == "utf-8":
            self.act_encoding_utf8.setChecked(True)
        elif encoding == "cp932":
            self.act_encoding_cp932.setChecked(True)
        elif encoding == "mbcs":
            self.act_encoding_ansi.setChecked(True)

                "O layout foi resetado. Reinicie o programa para aplicar as mudanças."

    def _update_title(self):
        """Atualiza título da janela"""

    def _load_settings(self):
        """Carrega configurações salvas"""
        geometry = self.settings.value("geometry")
        if geometry:
            self.restoreGeometry(geometry)

        state = self.settings.value("windowState")
        if state:
            self.restoreState(state)

        encoding = self.settings.value("encoding", "auto")
        self._set_encoding(encoding)

        snap_enabled = self.settings.value("snap_enabled", False, type=bool)
        self.act_snap.setChecked(snap_enabled)
        self.waveform.set_snap_enabled(snap_enabled)

        snap_mode = self.settings.value("snap_mode", "peaks")
        if snap_mode == "peaks":
            self.act_snap_peaks.setChecked(True)
        elif snap_mode == "zero_crossing":
            self.act_snap_zero_crossing.setChecked(True)
        else:
            self.act_snap_none.setChecked(True)
        self.waveform.set_snap_mode(snap_mode)

        wave_color_scheme = self.settings.value("wave_color", "blue")
        if wave_color_scheme == "green":
            self.act_wave_green.setChecked(True)
            self.waveform.set_wave_colors("#2dff88", "#c9ffe0")
        elif wave_color_scheme == "mono":
            self.act_wave_mono.setChecked(True)
            self.waveform.set_wave_colors("#ffffff", "#000000")
        else:
            self.act_wave_blue.setChecked(True)

        show_minimap = self.settings.value("show_minimap", True, type=bool)
        self.act_show_minimap.setChecked(show_minimap)
        self.waveform.set_show_minimap(show_minimap)

        auto_save = self.settings.value("auto_save", False, type=bool)
        self.act_toggle_auto_save.setChecked(auto_save)
        if auto_save:
            self._toggle_auto_save(True)

        srp_enabled = self.settings.value("srp_enabled", False, type=bool)
        self.act_toggle_srp.setChecked(srp_enabled)
        self._toggle_srp(srp_enabled)

        persistent_zoom = self.settings.value("persistent_zoom", True, type=bool)
        self.act_persistent_zoom.setChecked(persistent_zoom)
        self._toggle_persistent_zoom(persistent_zoom)

        if GPU_BACKEND_AVAILABLE:
            gpu_enabled_setting = self.settings.value("gpu_enabled", False, type=bool)
            if gpu_enabled_setting and gpu_available():
                self.act_toggle_gpu.setChecked(True)
                self._toggle_gpu_acceleration(True)

    def _save_settings(self):
        """Salva configurações"""
        self.settings.setValue("geometry", self.saveGeometry())
        self.settings.setValue("windowState", self.saveState())
        self.settings.setValue("encoding", self._encoding)
        self.settings.setValue("snap_enabled", self.act_snap.isChecked())
        self.settings.setValue("snap_mode", self.waveform.get_snap_mode())

        if self.act_wave_green.isChecked():
            self.settings.setValue("wave_color", "green")
        elif self.act_wave_mono.isChecked():
            self.settings.setValue("wave_color", "mono")
        else:
            self.settings.setValue("wave_color", "blue")

        self.settings.setValue("show_minimap", self.act_show_minimap.isChecked())
        self.settings.setValue("auto_save", self.act_toggle_auto_save.isChecked())
        self.settings.setValue("srp_enabled", self.act_toggle_srp.isChecked())
        self.settings.setValue("persistent_zoom", self.act_persistent_zoom.isChecked())

        if GPU_BACKEND_AVAILABLE:
            self.settings.setValue("gpu_enabled", self.act_toggle_gpu.isChecked())

    def closeEvent(self, event: QCloseEvent):
        """Manipula o fechamento da janela"""
        if self._dirty:
            completed = len(self._completed_aliases)
            total = self.table.rowCount()
            last_saved = self._get_last_saved_string()

            dialog = AdvancedExitDialog(
                self,
                self._current_path or Path("sem_arquivo.ini"),
                self._dirty,
                completed,
                total,
                last_saved
            )
            dialog.exec()
            action = dialog.get_action()

            if action == "cancel":
                event.ignore()
                return
            elif action == "backup":
                if self._create_backup():
                    self.save_oto()
                else:
                    reply = QMessageBox.question(
                        self, "Backup Falhou",
                        "Não foi possível criar backup. Salvar mesmo assim?",
                        QMessageBox.StandardButton.Yes | QMessageBox.StandardButton.No
                    )
                    if reply == QMessageBox.StandardButton.Yes:
                        self.save_oto()
                    else:
                        event.ignore()
                        return
            elif action == "save":
                self.save_oto()

        self._save_settings()
        event.accept()
