# audio_loader.py

from __future__ import annotations

import wave
from pathlib import Path
from typing import Tuple

import numpy as np
from PySide6.QtCore import QThread, Signal

try:
    from backend_gpu import get_gpu_backend, gpu_enabled
    GPU_BACKEND_AVAILABLE = True
except ImportError:
    GPU_BACKEND_AVAILABLE = False

    def gpu_enabled():
        return False


class WaveformLoaderThread(QThread):
    """
    Thread para carregar e processar a forma de onda de um arquivo WAV
    de forma assíncrona.
    """
    finished = Signal(np.ndarray, np.ndarray, str)  # (times, values, cache_key)

    def __init__(self, path: Path, max_points: int, cache_key: str):
        super().__init__()
        self.path = path
        self.max_points = max_points
        self.cache_key = cache_key

    def run(self):
        """
        Executa o carregamento e processamento do áudio na thread secundária.
        Emite o sinal 'finished' com os dados processados ou arrays vazios em caso de erro.
        """
        try:
            with wave.open(str(self.path), "rb") as wf:
                n_channels = wf.getnchannels()
                n_frames = wf.getnframes()
                sampwidth = wf.getsampwidth()
                framerate = wf.getframerate()
                raw = wf.readframes(n_frames)

            if GPU_BACKEND_AVAILABLE and gpu_enabled():
                backend = get_gpu_backend()
                times, values = backend.process_audio(
                    raw, n_channels, sampwidth, framerate, self.max_points
                )
            else:
                times, values = self._process_cpu(
                    raw, n_channels, sampwidth, framerate
                )

            self.finished.emit(times, values, self.cache_key)

        except Exception as e:
            print(f"Erro ao carregar waveform: {e}")
            # Emite arrays vazios em caso de erro
            self.finished.emit(
                np.array([], dtype=np.float32),
                np.array([], dtype=np.float32),
                self.cache_key
            )

    def _process_cpu(self, raw: bytes, n_channels: int, sampwidth: int, framerate: int) -> Tuple[np.ndarray, np.ndarray]:
        """
        Processa os dados de áudio brutos na CPU.
        Retorna arrays de tempo e valores normalizados.
        """
        if sampwidth == 1:
            data = np.frombuffer(raw, dtype=np.uint8)
            data = (data.astype(np.float32) - 128.0) / 128.0
        elif sampwidth == 2:
            data = np.frombuffer(raw, dtype=np.int16)
            data = data.astype(np.float32) / 32768.0
        else:
            # Tratamento para outros sampwidths, normalizando pelo valor máximo absoluto
            data = np.frombuffer(raw, dtype=np.int16).astype(np.float32)
            max_val = np.max(np.abs(data))
            if max_val > 0:
                data /= max_val

        if n_channels > 1:
            # Converte para mono calculando a média dos canais
            data = data.reshape(-1, n_channels).mean(axis=1)

        total = data.size
        if total == 0 or framerate <= 0:
            return np.array([], dtype=np.float32), np.array([], dtype=np.float32)

        # Downsampling por decimação (pega um ponto a cada 'step')
        step = max(1, total // self.max_points)
        data_ds = data[::step]
        n = data_ds.size
        times = (np.arange(n, dtype=np.float32) * step) / float(framerate)

        # Normaliza os valores pelo valor máximo absoluto do segmento downsampled
        max_amp = np.max(np.abs(data_ds)) if n > 0 else 1.0
        if max_amp > 0:
            values = data_ds / max_amp
        else:
            values = data_ds # Se max_amp for 0, não normaliza

        return times, values


def load_waveform_sync(path: Path, max_points: int = 8000) -> Tuple[np.ndarray, np.ndarray]:
    """
    Função para carregar e processar a forma de onda de forma síncrona.
    Útil para operações que não podem ser feitas em thread (como cache inicial).
    """
    if not path.is_file():
        return np.array([], dtype=np.float32), np.array([], dtype=np.float32)

    try:
        with wave.open(str(path), "rb") as wf:
            n_channels = wf.getnchannels()
            n_frames = wf.getnframes()
            sampwidth = wf.getsampwidth()
            framerate = wf.getframerate()
            raw = wf.readframes(n_frames)
    except Exception as e:
        print(f"Erro ao abrir arquivo WAV: {e}")
        return np.array([], dtype=np.float32), np.array([], dtype=np.float32)

    # Reutiliza a lógica de processamento da thread
    if GPU_BACKEND_AVAILABLE and gpu_enabled():
        backend = get_gpu_backend()
        return backend.process_audio(
            raw, n_channels, sampwidth, framerate, max_points
        )
    else:
        loader_thread_instance = WaveformLoaderThread(path, max_points, "")
        return loader_thread_instance._process_cpu(raw, n_channels, sampwidth, framerate)
