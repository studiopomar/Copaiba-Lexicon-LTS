# spectrogram_widget.py - VERSÃO COM GPU E CONFIGURÁVEL

from __future__ import annotations

import numpy as np
from PySide6.QtWidgets import QWidget, QVBoxLayout, QSizePolicy
from PySide6.QtCore import Qt, Signal
import pyqtgraph as pg

# Tenta importar GPU
try:
    from backend_gpu import gpu_enabled, get_gpu_backend

    GPU_AVAILABLE = True
except ImportError:
    GPU_AVAILABLE = False


    def gpu_enabled():
        return False


class SpectrogramWidget(QWidget):
    """
    Widget otimizado para exibir espectrograma do áudio.
    Suporta GPU e configurações ajustáveis.
    """

    # Colormaps disponíveis
    COLORMAPS = {
        'viridis': [
            (0, 0, 0), (0, 0, 128), (0, 100, 80), (0, 200, 100),
            (128, 255, 0), (255, 255, 0), (255, 200, 0), (255, 100, 0)
        ],
        'magma': [
            (0, 0, 0), (30, 0, 50), (80, 0, 80), (150, 20, 60),
            (200, 80, 20), (255, 180, 50), (255, 255, 200), (255, 255, 255)
        ],
        'hot': [
            (0, 0, 0), (64, 0, 0), (128, 0, 0), (255, 0, 0),
            (255, 128, 0), (255, 255, 0), (255, 255, 128), (255, 255, 255)
        ],
        'cool': [
            (0, 0, 0), (0, 0, 128), (0, 128, 255), (0, 255, 255),
            (128, 255, 255), (200, 255, 255), (220, 255, 255), (255, 255, 255)
        ],
        'gray': [
            (0, 0, 0), (32, 32, 32), (64, 64, 64), (96, 96, 96),
            (128, 128, 128), (160, 160, 160), (192, 192, 192), (255, 255, 255)
        ]
    }

    def __init__(self, parent=None):
        super().__init__(parent)

        self._wave_data = None
        self._sample_rate = 44100
        self._audio_duration = 0.0

        # Parâmetros configuráveis
        self._window_size = 2048
        self._hop_size = 512
        self._n_fft = 2048
        self._max_freq = 4000
        self._min_freq = 0
        self._gamma = 0.5  # Controle de intensidade
        self._contrast = 1.0  # Controle de contraste
        self._colormap_name = 'viridis'
        self._resolution_quality = 'medium'  # 'low', 'medium', 'high'
        self._use_gpu = False

        # Cache
        self._spectrogram_cache = None
        self._cache_valid = False

        # Plot widget
        self._plot = pg.PlotWidget(self)
        self._plot.setMenuEnabled(False)
        self._plot.showGrid(x=False, y=False)
        self._plot.setMouseEnabled(x=True, y=False)
        self._plot.setBackground('k')

        # Configurações dos eixos
        self._plot.setLabel('left', 'Frequency', units='Hz')
        self._plot.setLabel('bottom', 'Time', units='s')
        self._plot.getAxis('left').setTextPen('w')
        self._plot.getAxis('bottom').setTextPen('w')

        # ImageItem
        self._img_item = pg.ImageItem()
        self._plot.addItem(self._img_item)

        # Colormap
        self._setup_colormap()

        # Layout
        layout = QVBoxLayout(self)
        layout.setContentsMargins(0, 0, 0, 0)
        layout.setSpacing(0)
        layout.addWidget(self._plot)

        self.setFixedHeight(150)
        self.setSizePolicy(QSizePolicy.Expanding, QSizePolicy.Fixed)

        self._needs_update = False

    def _setup_colormap(self):
        """Configura colormap baseado na seleção"""
        colors = self.COLORMAPS.get(self._colormap_name, self.COLORMAPS['viridis'])

        lut = []
        n_colors = len(colors)

        for i in range(256):
            pos = i / 255.0 * (n_colors - 1)
            idx = int(pos)
            frac = pos - idx

            if idx >= n_colors - 1:
                color = colors[-1]
            else:
                c1 = np.array(colors[idx])
                c2 = np.array(colors[idx + 1])
                color = (c1 * (1 - frac) + c2 * frac).astype(int)

            lut.append(color)

        self._colormap = np.array(lut, dtype=np.ubyte)
        self._img_item.setLookupTable(self._colormap)

    def set_colormap(self, name: str):
        """Define o colormap"""
        if name in self.COLORMAPS:
            self._colormap_name = name
            self._setup_colormap()
            if self._cache_valid:
                self._update_display_full()

    def set_gamma(self, gamma: float):
        """Define o gamma (intensidade): 0.1 a 2.0"""
        self._gamma = max(0.1, min(2.0, gamma))
        self._cache_valid = False
        if self._wave_data is not None and self.isVisible():
            self._compute_spectrogram_cached()

    def set_contrast(self, contrast: float):
        """Define o contraste: 0.5 a 2.0"""
        self._contrast = max(0.5, min(2.0, contrast))
        self._cache_valid = False
        if self._wave_data is not None and self.isVisible():
            self._compute_spectrogram_cached()

    def set_freq_range(self, min_freq: int, max_freq: int):
        """Define o range de frequência"""
        self._min_freq = max(0, min_freq)
        self._max_freq = min(max_freq, self._sample_rate // 2)
        self._cache_valid = False
        if self._wave_data is not None and self.isVisible():
            self._compute_spectrogram_cached()

    def set_resolution_quality(self, quality: str):
        """Define qualidade: 'low', 'medium', 'high'"""
        if quality in ['low', 'medium', 'high']:
            self._resolution_quality = quality

            # Ajusta parâmetros baseado na qualidade
            if quality == 'low':
                self._window_size = 1024
                self._hop_size = 512
            elif quality == 'medium':
                self._window_size = 2048
                self._hop_size = 512
            else:  # high
                self._window_size = 4096
                self._hop_size = 256

            self._cache_valid = False
            if self._wave_data is not None and self.isVisible():
                self._compute_spectrogram_cached()

    def set_use_gpu(self, use_gpu: bool):
        """Ativa/desativa uso de GPU"""
        self._use_gpu = use_gpu and GPU_AVAILABLE and gpu_enabled()
        self._cache_valid = False

    def set_audio_data(self, wave_data: np.ndarray, sample_rate: int, duration: float):
        """Define os dados de áudio"""
        self._wave_data = wave_data
        self._sample_rate = sample_rate
        self._audio_duration = duration
        self._cache_valid = False
        self._needs_update = True

        if self.isVisible():
            self._compute_spectrogram_cached()

    def _compute_spectrogram_gpu(self, data: np.ndarray, window: np.ndarray,
                                 hop_size: int, n_fft: int) -> np.ndarray:
        """Calcula espectrograma usando GPU"""
        try:
            backend = get_gpu_backend()
            xp = backend.get_array_module()

            # Transfere dados para GPU
            data_gpu = xp.asarray(data)
            window_gpu = xp.asarray(window)

            window_size = len(window)
            n_frames = (len(data) - window_size) // hop_size + 1

            # Limita frames
            max_frames = 3000 if self._resolution_quality == 'high' else 2000
            if n_frames > max_frames:
                frame_step = n_frames // max_frames
                n_frames = max_frames
            else:
                frame_step = 1

            # Frequências
            n_freq_bins = n_fft // 2 + 1
            max_freq_bin = int(self._max_freq * n_fft / self._sample_rate)
            min_freq_bin = int(self._min_freq * n_fft / self._sample_rate)
            max_freq_bin = min(max_freq_bin, n_freq_bins)
            n_bins = max_freq_bin - min_freq_bin

            # Matriz do espectrograma na GPU
            spectrogram_gpu = xp.zeros((n_bins, n_frames), dtype=xp.float32)

            for i in range(n_frames):
                start = i * hop_size * frame_step
                end = start + window_size

                if end > len(data_gpu):
                    break

                # Janela e FFT na GPU
                windowed = data_gpu[start:end] * window_gpu
                fft_result = xp.fft.rfft(windowed, n=n_fft)[min_freq_bin:max_freq_bin]

                # Power spectrum
                power = xp.abs(fft_result) ** 2
                power = xp.maximum(power, 1e-10)
                spectrogram_gpu[:, i] = 10 * xp.log10(power)

            # Transfere de volta para CPU
            spectrogram = xp.asnumpy(spectrogram_gpu)

            return spectrogram

        except Exception as e:
            print(f"Erro ao usar GPU, fallback para CPU: {e}")
            return None

    def _compute_spectrogram_cached(self):
        """Calcula o espectrograma com cache"""
        if self._cache_valid:
            return

        if self._wave_data is None or len(self._wave_data) == 0:
            return

        data = self._wave_data

        # Converte para mono
        if len(data.shape) > 1:
            data = np.mean(data, axis=1)

        # Downsample para melhor performance
        target_rate = 22050 if self._resolution_quality == 'high' else 16000
        if self._sample_rate > target_rate * 1.2:
            downsample_factor = self._sample_rate // target_rate
            data = data[::downsample_factor]
            actual_rate = self._sample_rate / downsample_factor
        else:
            actual_rate = self._sample_rate

        # Atualiza sample rate interno
        original_sr = self._sample_rate
        self._sample_rate = actual_rate

        # Parâmetros
        window_size = min(self._window_size, len(data) // 4)
        hop_size = self._hop_size
        n_fft = self._n_fft

        # Janela Hann
        window = np.hanning(window_size)

        # Tenta GPU primeiro
        spectrogram = None
        if self._use_gpu and GPU_AVAILABLE and gpu_enabled():
            spectrogram = self._compute_spectrogram_gpu(data, window, hop_size, n_fft)

        # Fallback para CPU
        if spectrogram is None:
            n_frames = (len(data) - window_size) // hop_size + 1

            # Limita frames baseado na qualidade
            max_frames = {
                'low': 1000,
                'medium': 2000,
                'high': 3000
            }.get(self._resolution_quality, 2000)

            if n_frames > max_frames:
                frame_step = n_frames // max_frames
                n_frames = max_frames
            else:
                frame_step = 1

            # Frequências
            n_freq_bins = n_fft // 2 + 1
            max_freq_bin = int(self._max_freq * n_fft / actual_rate)
            min_freq_bin = int(self._min_freq * n_fft / actual_rate)
            max_freq_bin = min(max_freq_bin, n_freq_bins)
            n_bins = max_freq_bin - min_freq_bin

            # Matriz do espectrograma
            spectrogram = np.zeros((n_bins, n_frames), dtype=np.float32)

            for i in range(n_frames):
                start = i * hop_size * frame_step
                end = start + window_size

                if end > len(data):
                    break

                windowed = data[start:end] * window
                fft_result = np.fft.rfft(windowed, n=n_fft)[min_freq_bin:max_freq_bin]

                # Power spectrum em dB
                power = np.abs(fft_result) ** 2
                power = np.maximum(power, 1e-10)
                spectrogram[:, i] = 10 * np.log10(power)

        # Restaura sample rate original
        self._sample_rate = original_sr

        # Normalização com contraste ajustável
        vmin = np.percentile(spectrogram, 1)
        vmax = np.percentile(spectrogram, 99)

        # Aplica contraste
        vrange = (vmax - vmin) / self._contrast
        vmin = vmin + (vmax - vmin - vrange) / 2
        vmax = vmin + vrange

        if vmax > vmin:
            spectrogram = (spectrogram - vmin) / (vmax - vmin)
        else:
            spectrogram = np.zeros_like(spectrogram)

        spectrogram = np.clip(spectrogram, 0, 1)

        # Correção gamma (intensidade)
        spectrogram = np.power(spectrogram, self._gamma)

        # Converte para uint8
        self._spectrogram_cache = (spectrogram * 255).astype(np.uint8)
        self._cache_valid = True

        # Atualiza display
        self._update_display_full()

    def _update_display_full(self):
        """Atualiza a exibição"""
        if self._spectrogram_cache is None:
            return

        # Transpõe
        spec_display = self._spectrogram_cache.T

        # Define a imagem
        self._img_item.setImage(
            spec_display,
            levels=(0, 255),
            autoLevels=False
        )

        # Limites
        self._plot.setLimits(
            xMin=0, xMax=self._audio_duration,
            yMin=self._min_freq, yMax=self._max_freq
        )

        # Transformação
        if self._spectrogram_cache.shape[1] > 0 and self._spectrogram_cache.shape[0] > 0:
            x_scale = self._audio_duration / self._spectrogram_cache.shape[1]
            y_scale = (self._max_freq - self._min_freq) / self._spectrogram_cache.shape[0]

            tr = pg.QtGui.QTransform()
            tr.scale(x_scale, y_scale)
            tr.translate(0, self._min_freq / y_scale)
            self._img_item.setTransform(tr)

        # Range inicial
        self._plot.setRange(
            xRange=(0, self._audio_duration),
            yRange=(self._min_freq, self._max_freq),
            padding=0
        )

    def set_visible_region(self, start_time: float, end_time: float):
        """Atualiza a região visível"""
        if not self._cache_valid and self._needs_update:
            self._compute_spectrogram_cached()

        if self._cache_valid:
            self._plot.setXRange(start_time, end_time, padding=0)

    def showEvent(self, event):
        """Calcula quando visível"""
        super().showEvent(event)
        if self._needs_update and not self._cache_valid:
            self._compute_spectrogram_cached()

    def clear(self):
        """Limpa o espectrograma"""
        self._wave_data = None
        self._spectrogram_cache = None
        self._cache_valid = False
        self._needs_update = False
        self._img_item.clear()

    def set_height(self, height: int):
        """Define a altura do widget"""
        self.setFixedHeight(max(100, min(height, 300)))
